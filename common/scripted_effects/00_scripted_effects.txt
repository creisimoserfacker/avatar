evaluate_AI_resource_priorities_country = {
	if = {
		limit = {
			has_country_flag = AIs_dont_evaluate_resources
		}
		break = yes
	}

	if = {
		limit = {
			check_variable = {
				THIS.resource@food_resource < 1
			}
		}
		set_country_flag = AI_focus_on_food
		else_if = {
			limit = { 
				check_variable = {
					THIS.resource@aluminium < 1
				}
			}
			set_country_flag = AI_focus_on_wood
		}
		else_if = {
			limit = { 
				check_variable = {
					THIS.resource@chromium < 1
				}
			}
			set_country_flag = AI_focus_on_animal_products
		}
	}

}
evaluate_AI_resource_priorities = {
	every_country = {
		limit = {
			has_country_flag = AI_focus_on_food
		}
		clr_country_flag = AI_focus_on_food
	}
	every_country = {
		limit = {
			has_country_flag = AI_focus_on_wood
		}
		clr_country_flag = AI_focus_on_wood
	}
	every_country = {
		limit = {
			has_country_flag = AI_focus_on_animal_products
		}
		clr_country_flag = AI_focus_on_animal_products
	}
	every_country = {
		limit = {
			is_ai = yes
		}
		evaluate_AI_resource_priorities_country = yes
	}
}
calculate_earth_kingdom_animal_products_total = {
    if = {
        limit = {
            check_variable = {
                var = animal_products_in_earth_kingdom
                value = 0
                compare = not_equals 
            }
        }
        set_variable = {animal_products_in_earth_kingdom = 0}
    }
    log = "Animal products variable set to [?animal_products_in_earth_kingdom]"
    every_country = {
        limit = {
            is_in_earth_kingdom = yes
        }
        add_to_variable = {
            var = PREV.animal_products_in_earth_kingdom
            value = this.resource_produced@chromium
        }
            log = "Animal products in [THIS.GetNameDef] [?this.resource_produced@chromium] running total: [?PREV.animal_products_in_earth_kingdom]"
    }
    log = "Animal products in earth kingdom [?animal_products_in_earth_kingdom]"
}
